8.8 DIAGRAMA DE SEQUENCIA - REPRODUCAO DE GRAVACAO
Arquivo sugerido: diagrama-sequencia-reproducao-gravacao.png
Legenda: DIAGRAMA DE SEQUENCIA DE REPRODUCAO DE GRAVACAO
Onde inserir: Capitulo 6, nova subsecao de diagramas de sequencia
================================================================================

@startuml
skinparam sequenceMessageAlign center

actor Usuario as U
participant "MinhasGravacoesPage" as MG
participant "ContextualVoiceMixin" as MIXIN
participant "CommandService" as CS
participant "RecordingsListController" as CTRL
participant "AudioPlayerService" as APS
participant "VoiceSessionManager" as VSM
participant "HistoricoRepository" as HIST
participant "AppDatabase\n(SQLite)" as DB

U -> MG: Acessa Minhas Gravacoes
MG -> CTRL: carregarGravacoes(usuarioId)
CTRL -> DB: SELECT FROM gravacao WHERE usuario_id=?
DB --> CTRL: lista de gravacoes
CTRL --> MG: exibir lista com indices visuais (Item 1, Item 2...)

U -> MG: Diz "reproduzir item 2"
MG -> MIXIN: onVoiceResult("reproduzir item 2")
MIXIN -> CS: interpret("reproduzir item 2")
CS --> MIXIN: CommandResult(type=reproduzirGravacao, parametro="item 2")

MIXIN -> CTRL: resolverIndiceVisual("item 2")
CTRL -> CTRL: obter gravacao na posicao 2 da lista
CTRL --> MIXIN: Gravacao (id, nome, caminhoArquivo)

MIXIN -> VSM: verificar estado (nao deve ser recordingLocked)
VSM --> MIXIN: estado=listening (ok para reproduzir)

MIXIN -> APS: iniciarReproducao(caminhoArquivo)

alt Arquivo existe no dispositivo
  APS --> MIXIN: reproducao iniciada
  MG -> U: Exibir indicador de reproducao ativa
  MG -> VSM: transitionTo(idle) [escuta suspensa durante reproducao]

  APS -> APS: reproduzir audio...

  APS -> MIXIN: onPlaybackComplete()
  MIXIN -> VSM: transitionTo(listening)
  MG -> U: Indicador de reproducao finalizado

  MIXIN -> HIST: registrar("gravacao_reproduzida", nome, gravacaoId)
  HIST -> DB: INSERT INTO historico_acao
else Arquivo ausente no dispositivo
  APS --> MIXIN: erro: arquivo nao encontrado
  MIXIN -> CTRL: marcarComoAusenteSeNecessario(gravacaoId)
  CTRL -> DB: UPDATE gravacao SET status='arquivo_ausente'
  MG -> U: Exibir mensagem de erro (arquivo nao encontrado)
end

alt Usuario diz "parar reproducao" durante reproducao
  U -> MG: Diz "parar"
  MG -> MIXIN: onVoiceResult("parar")
  MIXIN -> CS: interpret("parar")
  CS --> MIXIN: CommandResult(type=pararReproducao)
  MIXIN -> APS: pararReproducao()
  APS --> MIXIN: reproducao parada
  MIXIN -> VSM: transitionTo(listening)
end
@enduml

8.9 DIAGRAMA DE SEQUENCIA - RENOMEAR GRAVACAO POR VOZ
Arquivo sugerido: diagrama-sequencia-renomear-gravacao.png
Legenda: DIAGRAMA DE SEQUENCIA DE RENOMEACAO DE GRAVACAO POR VOZ
Onde inserir: Capitulo 6, nova subsecao de diagramas de sequencia
================================================================================

@startuml
skinparam sequenceMessageAlign center

actor Usuario as U
participant "DetalhesGravacaoPage" as DG
participant "ContextualVoiceMixin" as MIXIN
participant "CommandService" as CS
participant "VoiceConfirmationController" as VCC
participant "GravacaoRepository" as REPO
participant "HistoricoRepository" as HIST
participant "AppDatabase\n(SQLite)" as DB

U -> DG: Acessa Detalhes da Gravacao
DG -> U: Exibir nome atual da gravacao

U -> DG: Diz "renomear para nova melodia"
DG -> MIXIN: onVoiceResult("renomear para nova melodia")
MIXIN -> CS: interpret("renomear para nova melodia")
CS -> CS: _extractAfterAny(texto, ["renomear para", ...])
CS --> MIXIN: CommandResult(type=renomearGravacao, parametroSecundario="nova melodia")

MIXIN -> DG: iniciar fluxo de renomeacao
DG -> VCC: solicitarConfirmacao("Renomear para: 'Nova melodia'?")
DG -> U: Exibir confirmacao: "Renomear para 'Nova melodia'? Diga 'confirmar' ou 'cancelar'"

alt Usuario confirma
  U -> DG: Diz "confirmar" ou "sim"
  DG -> MIXIN: onVoiceResult("confirmar")
  MIXIN -> CS: interpret("confirmar")
  CS --> MIXIN: CommandResult(type=confirmarAcao)
  MIXIN -> VCC: confirmar()
  VCC -> REPO: renomear(gravacaoId, "Nova melodia")
  REPO -> DB: UPDATE gravacao SET nome='Nova melodia' WHERE id=?
  DB --> REPO: atualizado
  REPO --> DG: sucesso
  DG -> HIST: registrar("gravacao_renomeada", "Renomeada para 'Nova melodia'", gravacaoId)
  HIST -> DB: INSERT INTO historico_acao
  DG -> U: Exibir novo nome da gravacao
else Usuario cancela
  U -> DG: Diz "cancelar" ou "nao"
  DG -> MIXIN: onVoiceResult("cancelar")
  MIXIN -> CS: interpret("cancelar")
  CS --> MIXIN: CommandResult(type=cancelarAcao)
  MIXIN -> VCC: cancelar()
  DG -> U: Manter nome original
end
@enduml

8.10 DIAGRAMA DE SEQUENCIA - COMANDO PERSONALIZADO POR VOZ
Arquivo sugerido: diagrama-sequencia-comando-personalizado.png
Legenda: DIAGRAMA DE SEQUENCIA DE COMANDO PERSONALIZADO
Onde inserir: Capitulo 6, nova subsecao de diagramas de sequencia
================================================================================

@startuml
skinparam sequenceMessageAlign center

actor Usuario as U
participant "ConfiguracoesPage" as CFG
participant "ContextualVoiceMixin" as MIXIN
participant "CommandService" as CS
participant "CustomCommandService" as CCS
participant "CustomCommandCatalog" as CAT
participant "ComandoPersonalizadoRepository" as REPO
participant "AppDatabase\n(SQLite)" as DB

group Criacao do comando personalizado
  U -> CFG: Acessa secao Comandos Personalizados
  CFG -> U: Exibir formulario de novo comando

  U -> CFG: Digita ou diz a frase: "minha musica"
  CFG -> CCS: validarFrase("minha musica")
  CCS -> CCS: normalizar frase
  CCS -> CS: verificarConflito("minha musica")
  CS -> CS: interpret("minha musica")
  CS --> CCS: CommandResult(type=desconhecido) [sem conflito]
  CCS --> CFG: frase valida

  U -> CFG: Selecionar acao: "Abrir gravacoes"
  CFG -> CAT: listar acoes disponiveis
  CAT --> CFG: [abrir_gravacoes, abrir_projetos, abrir_editor, ...]
  U -> CFG: Escolhe "abrir_gravacoes"

  U -> CFG: Toca "Salvar"
  CFG -> CCS: salvar(usuarioId, "minha musica", "abrir_gravacoes")
  CCS -> REPO: inserir(ComandoPersonalizado)
  REPO -> DB: INSERT INTO comando_personalizado (usuario_id, frase, tipo_comando, ativo=1)
  DB --> REPO: id gerado
  REPO --> CFG: sucesso
  CFG -> U: Exibir "Comando 'minha musica' criado com sucesso"
end

group Execucao do comando personalizado
  U -> CFG: Diz "minha musica" em qualquer tela
  MIXIN -> CS: interpret("minha musica")
  CS --> MIXIN: CommandResult(type=desconhecido) [nao reconhecido como global]
  MIXIN -> CCS: buscarComandoPersonalizado(usuarioId, "minha musica")
  CCS -> REPO: findByFrase(usuarioId, "minha musica")
  REPO -> DB: SELECT FROM comando_personalizado WHERE usuario_id=? AND frase=? AND ativo=1
  DB --> REPO: ComandoPersonalizado(tipo_comando="abrir_gravacoes")
  REPO --> MIXIN: ComandoPersonalizado encontrado
  MIXIN -> MIXIN: criar CommandResult(type=abrirGravacoes)
  MIXIN -> MIXIN: dispatch(result)
  MIXIN -> U: Navegar para Minhas Gravacoes
end
@enduml

8.11 DIAGRAMA DE SEQUENCIA - PARADA AUTOMATICA POR SILENCIO
Arquivo sugerido: diagrama-sequencia-parada-silencio.png
Legenda: DIAGRAMA DE SEQUENCIA DE PARADA AUTOMATICA POR SILENCIO
Onde inserir: Capitulo 6, nova subsecao de diagramas de sequencia
================================================================================

@startuml
skinparam sequenceMessageAlign center

actor Usuario as U
participant "EditorPage" as EP
participant "RecordingRealtimeCoordinator" as RRC
participant "ConfiguracaoAppRepository" as CFG_REPO
participant "AudioRecordingService" as ARS
participant "AppDatabase\n(SQLite)" as DB

U -> EP: Acessa o Editor
EP -> CFG_REPO: lerConfiguracoes()
CFG_REPO -> DB: SELECT FROM configuracao_app WHERE id=1
DB --> CFG_REPO: {parada_silencio=1, tempo_silencio_segundos=3, limiar_silencio_db=-30.0}
CFG_REPO --> EP: ConfiguracaoApp

U -> EP: Diz "gravar"
EP -> RRC: iniciarGravacao()
RRC -> ARS: startRecording()
ARS --> RRC: gravacao ativa
RRC -> RRC: silenceMs = 0, lastActiveMs = now
RRC --> EP: RecordingRealtimeState(recording=true)

loop Ciclo de amplitude (a cada ~100ms)
  ARS -> RRC: onAmplitude(dBvalue)
  RRC -> EP: atualizar barra de amplitude
  EP -> U: Mostrar nivel de audio

  alt Amplitude acima do limiar (-30 dB)
    RRC -> RRC: silenceMs = 0
    RRC -> RRC: lastActiveMs = now
  else Amplitude abaixo do limiar (silencio detectado)
    RRC -> RRC: silenceMs += delta
    EP -> U: Indicador de silencio visivel

    alt silenceMs >= 3000ms (3 segundos de silencio)
      RRC -> RRC: parada automatica acionada
      RRC -> ARS: stopRecording()
      ARS --> RRC: path, duracao, tamanho
      RRC -> DB: INSERT INTO gravacao (status='concluida', ...)
      DB --> RRC: gravacao salva
      RRC -> DB: INSERT INTO historico_acao (tipo='gravacao_automatica_por_silencio')
      RRC --> EP: RecordingRealtimeState(recording=false)
      EP -> U: Exibir "Gravacao parada automaticamente por silencio"
    end
  end
end

note over RRC
  O ciclo so executa se parada_silencio=1
  nas configuracoes. Se desativado, apenas
  exibe amplitude sem encerrar automaticamente.
end note
@enduml

8.14 DIAGRAMA DE ESTADOS DA ESCUTA DE VOZ
Arquivo sugerido: diagrama-estados-voz.png
Legenda: DIAGRAMA DE ESTADOS DA ESCUTA DE VOZ DO TOUCHLESS
Onde inserir: Capitulo 6, nova secao DIAGRAMA DE ESTADOS DA ESCUTA DE VOZ
================================================================================

@startuml
skinparam state {
  BackgroundColor LightYellow
  BorderColor DarkOrange
  FontSize 11
}

[*] --> idle : app inicializado

state idle {
  [*] --> aguardando
  aguardando : microfone inativo\nescuta pausada
}

idle --> listening : startListening()\nescuta continua ativa ou manual

state listening {
  [*] --> microfone_ativo
  microfone_ativo : STT ativo\naguardando fala do usuario
}

listening --> processingCommand : resultado de voz recebido\n(texto reconhecido)
listening --> error : falha no STT\ntimeout de reconhecimento
listening --> manualPaused : usuario pausa\nmanualmente
listening --> recordingLocked : gravacao iniciada\n(microfone em uso)

state processingCommand {
  [*] --> interpretando
  interpretando : CommandService.interpret()\ndespacho do comando
}

processingCommand --> idle : comando executado\nescuta continua desativada
processingCommand --> listening : comando executado\nescuta continua ativa
processingCommand --> aiThinking : comando requer\nprocessamento assincrono

state aiThinking {
  [*] --> aguardando_resultado
  aguardando_resultado : aguardando resposta\nassincrona
}

aiThinking --> idle : resultado processado\nescuta continua desativada
aiThinking --> listening : resultado processado\nescuta continua ativa
aiThinking --> error : timeout ou falha

state manualPaused {
  [*] --> escuta_pausada
  escuta_pausada : usuario pausou\nmanualmente
}

manualPaused --> listening : usuario reativa\nescuta
manualPaused --> idle : app perde foco\nou navega para fora

state recordingLocked {
  [*] --> microfone_em_uso
  microfone_em_uso : gravacao ativa\ncomandos de voz suspensos\nexceto os de gravacao
}

recordingLocked --> idle : gravacao encerrada\nescuta continua desativada
recordingLocked --> listening : gravacao encerrada\nescuta continua ativa

state error {
  [*] --> estado_erro
  estado_erro : falha no reconhecimento\nVoiceRecognitionErrorGuard ativo
}

error --> listening : recuperacao automatica\n(tentativa de restart)
error --> idle : max tentativas atingido\nou app em background

idle --> [*] : app encerrado ou\nusuario faz logout
@enduml

8.15 DIAGRAMA DE PRIORIDADE DE COMANDOS DE VOZ

@startuml
skinparam activityFontSize 11
skinparam activityBackgroundColor LightGreen
skinparam activityBorderColor DarkGreen
skinparam decisionBackgroundColor LightBlue

start

:Texto reconhecido pelo STT;

if (Modal critico esta aberto?) then (SIM)
  :Bloquear todos os comandos;
  :Exibir feedback: "Feche o dialogo primeiro";
  stop
else (NAO)
endif

if (Modo guiado esta ativo?) then (SIM)
  :Capturar entrada especifica do modo guiado;
  note right
    Ex: aguardando nome do projeto,
    nome da gravacao, descricao, etc.
  end note
  :Processar como entrada de modo guiado;
  stop
else (NAO)
endif

if (Dialogo de ajuda esta aberto?) then (SIM)
  :Aceitar apenas comando de fechar;
  :Ex: "cancelar", "fechar";
  stop
else (NAO)
endif

if (Gravacao ativa (recordingLocked)?) then (SIM)
  :Verificar se e comando de gravacao;
  if (pausar, retomar, finalizar, salvar?) then (SIM)
    :Executar comando de gravacao;
    stop
  else (NAO)
    :Ignorar comando;
    :Feedback: "Em gravacao";
    stop
  endif
else (NAO)
endif

if (Existe comando local da tela atual?) then (SIM)
  :Executar comando local da tela;
  note right
    Ex: "buscar gravacao por" em MinhasGravacoes,
    "nome do projeto" em NovoProjeto, etc.
  end note
  :Registrar em comando_voz;
  stop
else (NAO)
endif

if (Existe comando global de navegacao?) then (SIM)
  :Executar comando global;
  note right
    Ex: "meus projetos", "dashboard",
    "historico", "configuracoes", "voltar",
    "ajuda", "sair", etc.
  end note
  :Registrar em comando_voz;
  stop
else (NAO)
endif

if (Existe comando personalizado do usuario?) then (SIM)
  :Buscar em comando_personalizado WHERE ativo=1;
  :Executar acao mapeada;
  :Registrar em comando_voz;
  stop
else (NAO)
endif

:Fala nao reconhecida;
:Feedback: "Nao entendi. Diga: meus projetos,\nminhas gravacoes ou configuracoes.";
:Registrar em comando_voz (status='nao_reconhecido');

stop
@enduml
