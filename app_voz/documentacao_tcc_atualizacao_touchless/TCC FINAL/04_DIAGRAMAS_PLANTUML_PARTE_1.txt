@startuml
skinparam sequenceMessageAlign center
skinparam noteFontSize 10

actor Usuário as U
participant "AppEntryGate" as AEG
participant "WelcomePage" as WEL
participant "MicrophonePermPage" as MIC
participant "LoginPage" as LOG
participant "AuthService" as AUTH
participant "GoogleAuthService" as GAUTH
participant "AppDatabase\n(SQLite)" as DB
participant "HomePage" as HOME

U -> AEG: Abrir app
AEG -> DB: ler configuracao_app
DB --> AEG: primeira_execucao_concluida=false

AEG -> WEL: Navegar para Boas-vindas

U -> WEL: Toca "Começar" ou diz "começar"
WEL -> MIC: Navegar para Permissão de Microfone

U -> MIC: Toca "Permitir" ou diz "permitir microfone"
MIC -> MIC: Solicitar permissão ao sistema Android
MIC -> DB: atualizar primeira_execucao_concluida=true
MIC -> LOG: Navegar para Login

note over LOG: Usuário pode fazer login local ou com Google

alt Login Local
  U -> LOG: Informar e-mail + senha + toca "Entrar"
  LOG -> AUTH: autenticar(email, senha)
  AUTH -> DB: buscar usuario por email
  DB --> AUTH: usuario encontrado
  AUTH -> AUTH: verificar hash da senha
  AUTH --> LOG: sucesso
  LOG -> HOME: Navegar para Home
else Login Google
  U -> LOG: Tocar "Entrar com Google"
  LOG -> GAUTH: iniciarAutenticacaoGoogle()
  GAUTH -> GAUTH: Google OAuth flow
  GAUTH --> LOG: GoogleIdentity (email, googleId, nome, fotoUrl)
  LOG -> AUTH: autenticarComGoogle(identity)
  AUTH -> DB: buscar usuario por google_id
  alt Conta Google nova
    DB --> AUTH: não encontrado
    AUTH -> DB: criar usuario (auth_provider='google')
    DB --> AUTH: usuario criado
  else Conta Google existente
    DB --> AUTH: usuario encontrado
  end
  AUTH --> LOG: sucesso
  LOG -> HOME: Navegar para Home
else Usuário sem cadastro
  U -> LOG: Tocar "Criar conta"
  LOG -> LOG: Navegar para Cadastro
  U -> LOG: Informar nome, e-mail e senha
  LOG -> AUTH: cadastrar(nome, email, senha)
  AUTH -> AUTH: gerar salt + hash da senha
  AUTH -> DB: inserir usuario (auth_provider='local')
  DB --> AUTH: usuario criado
  AUTH --> LOG: sucesso
  LOG -> HOME: Navegar para Home
end

HOME -> U: Exibir tela inicial com comandos de voz ativos
@enduml

@startuml

participant "Tela Atual" as TELA
participant "ContextualVoiceListeningMixin" as MIXIN
participant "SpeechService\n(STT)" as STT
participant "CommandService" as CS
participant "VoiceCommandDispatcher" as VCD
participant "VoiceSessionManager" as VSM
participant "NavigationService" as NAV
participant "AppDatabase\n(SQLite)" as DB

U -> TELA: App em uso, escuta ativa
TELA -> MIXIN: voiceOuvindo = true
MIXIN -> VSM: transitionTo(listening)
MIXIN -> STT: iniciar reconhecimento

U -> STT: Fala "minhas gravações"
STT --> MIXIN: onResult("minhas gravações")

MIXIN -> MIXIN: aguarda coalescência (resultado final)
MIXIN -> CS: interpret("minhas gravações")
CS -> CS: normalize("minhas gravações") = "minhas gravacoes"
CS -> CS: _interpretEssentialCommand(...)
CS --> MIXIN: CommandResult(type=abrirGravacoes, recognized=true)

MIXIN -> VSM: transitionTo(processingCommand)
MIXIN -> VCD: dispatch(CommandResult, context)
VCD -> NAV: navegar para MinhasGravacoesPage
NAV --> TELA: nova tela exibida

VCD -> DB: inserir comando_voz (tipo='abrir_gravacoes', status='reconhecido')
VCD -> MIXIN: comando executado

MIXIN -> VSM: transitionTo(idle)

alt Escuta contínua ativa
  MIXIN -> STT: reiniciar reconhecimento
  MIXIN -> VSM: transitionTo(listening)
end

U -> TELA: Vê a tela de Minhas Gravações
@enduml

@startuml
skinparam sequenceMessageAlign center

actor Usuário as U
participant "EditorPage" as EP
participant "RecordingRealtimeCoordinator" as RRC
participant "AudioRecordingService" as ARS
participant "VoiceSessionManager" as VSM
participant "GravacaoRepository" as REPO
participant "HistoricoRepository" as HIST
participant "AppDatabase\n(SQLite)" as DB

U -> EP: Acessa o Editor (projeto selecionado)
EP -> RRC: inicializar coordenador
RRC -> VSM: verificar estado da sessão

U -> EP: Diz "gravar" ou toca botão Gravar
EP -> RRC: iniciarGravacao()
RRC -> VSM: transitionTo(recordingLocked)
RRC -> ARS: startRecording(path)
ARS --> RRC: gravação iniciada (startedAt = agora)
RRC --> EP: RecordingRealtimeState(recording=true)
EP -> U: Exibir indicador de gravação + amplitude

loop Monitoramento de amplitude
  ARS -> RRC: onAmplitude(dB)
  RRC -> EP: atualizar currentAmplitude
  EP -> U: Atualizar barra de nível de áudio

  alt Parada automática por silêncio ativa
    RRC -> RRC: silenceMs += delta
    alt silenceMs >= tempoSilencioMs E amplitude < limiarDb
      RRC -> RRC: encerrarGravacaoAutomatica()
    end
  end
end

alt Usuário encerra manualmente
  U -> EP: Diz "finalizar gravação" ou toca Parar
  EP -> RRC: encerrarGravacao(automatic=false)
else Parada automática
  RRC -> RRC: encerrarGravacao(automatic=true)
end

RRC -> ARS: stopRecording()
ARS --> RRC: path do arquivo, duração, tamanho

RRC -> REPO: finalizarGravacao(path, startedAt, automatic)
REPO -> DB: INSERT INTO gravacao (nome, caminho, status='concluida', formato_audio, tamanho_bytes)
DB --> REPO: gravacaoId
REPO --> RRC: Gravacao salva

RRC -> HIST: registrar("gravacao_finalizada", descricao, gravacaoId)
HIST -> DB: INSERT INTO historico_acao

RRC -> VSM: transitionTo(idle)
RRC --> EP: RecordingRealtimeState(recording=false)
EP -> U: Exibir "Gravação salva" + opção de reproduzir
@enduml
